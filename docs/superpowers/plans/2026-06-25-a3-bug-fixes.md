**Architecture:** All three fixes are in `batchbookui/` (the git submodule at `BatchBook/batchbookui`). Every change is isolated to one file per bug. After all fixes are committed inside the submodule, a final step bumps the submodule pointer in the parent BatchBook repo and opens a PR in each repo.

```jsx
const handleResendOtp = async () => {
  setError('');
  setResendTimer(60);
  setIsResending(true);

  try {
    const res = await fetch(`${import.meta.env.VITE_API_BASE_URL}/owner/generate_otp`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ phone: phoneNumber }),
    });

    if (!res.ok) {
      const data = await res.json().catch(() => ({}));
      throw new Error(data.detail || `Server error: ${res.status}`);
    }

    // OTP re-sent successfully — unblock inputs and clear old digits
    setOtp('');
    setIsResending(false);
  } catch (err) {
    setError('Failed to resend OTP. Please try again. ' + err.message);
    setIsResending(false);
  }
};
```
